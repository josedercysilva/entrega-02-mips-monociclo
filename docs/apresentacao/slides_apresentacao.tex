\begin{frame}{Visão Geral da Arquitetura MIPS Monociclo}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{Princípios Fundamentais do Processador}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item \textbf{Ciclo Único ($CPI = 1$):} Toda instrução é buscada, decodificada, executada e gravada no intervalo de exatamente um período de clock;
          \item \textbf{Largura de Dados e Endereçamento:} Barramentos de dados de 32 bits com memória endereçada por byte e alinhada por palavra (4 bytes);
          \item \textbf{Conjunto de Instruções Suportado:}
            \begin{itemize}
              \item \textit{Tipo-R:} \texttt{add}, \texttt{sub}, \texttt{and}, \texttt{or}, \texttt{slt}, \texttt{nor}, \texttt{jr};
              \item \textit{Tipo-I:} \texttt{addi}, \texttt{lw}, \texttt{sw}, \texttt{beq};
              \item \textit{Tipo-J:} \texttt{j}, \texttt{jal}.
            \end{itemize}
          \item \textbf{Sem Conflitos Estruturais:} Memórias fisicamente segregadas para instruções e dados (Arquitetura Harvard).
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Fluxo de Execução das Instruções}
        \footnotesize
        \begin{enumerate}\setlength{\itemsep}{3pt}
          \item \textbf{Busca (IF):} Leitura da ROM indexada por \texttt{PC[31:2]};
          \item \textbf{Decodificação (ID):} Acesso ao RegFile e geração dos sinais de controle;
          \item \textbf{Execução (EX):} Cálculo aritmético/lógico na ULA ou soma de endereço;
          \item \textbf{Memória (MEM):} Acesso à memória de dados (\texttt{lw} / \texttt{sw});
          \item \textbf{Escrita (WB):} Atualização do RegFile e comutação do próximo PC.
        \end{enumerate}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% Slide 4: ULA de 32 bits: Arquitetura e Complemento de Dois
% ------------------------------------------------------------------------------
\begin{frame}{ULA de 32 bits: Arquitetura e Otimização Lógica}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{Somador/Subtrator Compartilhado (Harris \& Harris)}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \textbf{Minimização de Área em Silício/FPGA:}
            Evita sintetizar blocos distintos de soma e subtração. Utiliza uma única cadeia somadora de 32 bits:
            $$B_{\text{inv}} = B \oplus \text{alu\_control}[2], \quad S = A + B_{\text{inv}} + \text{alu\_control}[2]$$
          \item Quando $\text{alu\_control}[2] = 1$, as portas XOR invertem os bits de $B$ e o bit atua como $C_{\text{in}} = 1$ (Complemento de Dois: $\overline{B} + 1$);
          \item \textbf{Geração da Flag Zero:} Redução unária $\text{flag\_zero} = \sim\!(|\,\text{alu\_result})$ via árvore balanceada de portas NOR ($O(1)$ lógico);
          \item \textbf{SLT Assinado:} Avalia o bit de sinal com tratamento estrito de overflow aritmético.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Mapeamento de Operações da ULA}
        \footnotesize
        \begin{center}
        \begin{tabular}{ccl}
          \toprule
          \textbf{alu\_control[2:0]} & \textbf{Função} & \textbf{Descrição} \\
          \midrule
          \texttt{000} & \texttt{AND} & Lógico bit a bit \\
          \texttt{001} & \texttt{OR}  & Lógico bit a bit \\
          \texttt{010} & \texttt{ADD} & Soma com sinal \\
          \texttt{110} & \texttt{SUB} & Subtração ($A - B$) \\
          \texttt{111} & \texttt{SLT} & Comparação com sinal \\
          \texttt{100} & \texttt{NOR} & Lógico bit a bit \\
          \bottomrule
        \end{tabular}
        \end{center}
        \vspace{1mm}
        \textbf{SLT Assinado:} Avalia o bit de sinal com correção estrita de overflow aritmético.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Banco de Registradores de 32 bits (\texttt{reg\_file.v})}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{Topologia Estrutural do Register File}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item \textbf{Capacidade:} 32 registradores de uso geral de 32 bits (\$0 a \$31);
          \item \textbf{Portas de Leitura (2 Portas):}
            \begin{itemize}
              \item Endereços: \texttt{raddr1[4:0]} e \texttt{raddr2[4:0]};
              \item Saídas de dados: \texttt{rdata1[31:0]} e \texttt{rdata2[31:0]};
              \item \textbf{Modo de Leitura:} Assíncrona / combinacional pura.
            \end{itemize}
          \item \textbf{Porta de Escrita (1 Porta):}
            \begin{itemize}
              \item Endereço de destino: \texttt{waddr[4:0]};
              \item Dado de entrada: \texttt{wdata[31:0]};
              \item Habilitação: \texttt{reg\_write} síncrono em \texttt{posedge clk}.
            \end{itemize}
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Garantia Estrutural do Registrador \$0}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{3pt}
          \item \textbf{Regra Canônica da Arquitetura MIPS:} O registrador \$0 armazena permanentemente a constante zero;
          \item \textbf{Amarração em Hardware (\textit{Hardwired Zero}):}
            \begin{center}
              \texttt{rdata = (raddr == 5'b0) ? 32'b0 : regs[raddr];}
            \end{center}
          \item Qualquer tentativa de escrita em \$0 é sumariamente ignorada pelo hardware, blindando a arquitetura contra desvios de software.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Unidade de Controle e Decodificação de Instruções}

      \begin{block}{Decodificação Hierárquica em Dois Níveis}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \textbf{Nível 1: Controle Principal (\texttt{Opcode[5:0]}):}
            Decodifica o formato da instrução e estabelece os sinais gerais de controle:
            \begin{itemize}\setlength{\itemsep}{1pt}
              \item \texttt{RegDst}, \texttt{ALUSrc}, \texttt{MemtoReg}, \texttt{RegWrite};
              \item \texttt{MemRead}, \texttt{MemWrite}, \texttt{Branch}, \texttt{Jump}, \texttt{Jal};
              \item Código de operação básica para a ULA: \texttt{ALUOp[1:0]}.
            \end{itemize}
          \item \textbf{Nível 2: ALU Control (\texttt{ALUOp[1:0]} + \texttt{Funct[5:0]}):}
            Define a operação exata da ULA (\texttt{alu\_control[2:0]}) e sinaliza retorno de procedimento (\texttt{JR\_Control}).
        \end{itemize}
      \end{block}
    \end{column}

          \bottomrule
        \end{tabular}

\begin{frame}{Decodificação dos Desvios: \texttt{beq}, \texttt{j}, \texttt{jal} e \texttt{jr}}
  \begin{block}{Matriz de Controle para Instruções de Salto}
    \scriptsize
    \setlength{\tabcolsep}{3pt}
    \centering
    \begin{tabular}{lcccccccc}

      \midrule
      \texttt{beq} & I & \textbf{1} & 0 & 0 & 0 & X & 0 & $PC+4$ ou $PC_{\text{branch}}$ \\

      \texttt{jal} & J & 0 & \textbf{1} & \textbf{1} & 0 & \textbf{2'b10} (\$31) & \textbf{1} & $\{PC+4[31:28], \text{instr}[25:0], 2'b00\}$ \\

      \bottomrule
    \end{tabular}
  \end{block}

  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Particularidades de \texttt{jal} (Call)}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item Salta para o endereço alvo do tipo J;
          \item Salva o endereço de retorno ($PC+4$) no registrador \texttt{\$31} (\texttt{\$ra});
          \item Multiplexador \texttt{mem\_to\_reg} seleciona $PC+4$.
        \end{itemize}
      \end{exampleblock}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Particularidades de \texttt{jr} (Return)}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item Identificado pelo código de função: \texttt{Funct = 0x08};
          \item Sinal \texttt{JR\_Control} comuta o multiplexador do PC;
          \item Restaura o fluxo para o endereço lido de \texttt{\$ra}.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

  \end{columns}

  \begin{exampleblock}{Vantagem da Separação Harvard no Processador Monociclo}
    \footnotesize
    A existência de barramentos físicos distintos para busca de código e acesso a dados permite que instruções de memória como \texttt{lw} realizem a busca da instrução e a leitura de dados no mesmo período de clock, eliminando conflitos de barramento.
  \end{exampleblock}
\end{frame}

% Slide 10: Fiação Estrutural do Top-Level e Próximo PC

\begin{frame}{Fiação Estrutural do Top-Level e Cascata do Próximo PC}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{block}{Cascata Ordenada de Multiplexadores do PC}
        \footnotesize
        A definição do próximo valor do Program Counter (\texttt{next\_pc}) obedece a uma hierarquia estrita de prioridades:
        \begin{enumerate}\setlength{\itemsep}{3pt}
          \item \textbf{Nível 1 (Branch Condicional):}
            $$PC_{\text{seq}} = (\text{branch} \land \text{zero}) \ ? \ PC_{\text{branch}} : PC+4$$
          \item \textbf{Nível 2 (Jump Incondicional / Call):}
            $$PC_{\text{j\_sel}} = \text{jump} \ ? \ \{PC+4[31:28], \text{instr}[25:0], 2'b00\} : PC_{\text{seq}}$$
          \item \textbf{Nível 3 (Return por Registrador -- Prioritário):}
            $$PC_{\text{next}} = \text{jr\_ctrl} \ ? \ \text{rdata1} : PC_{\text{j\_sel}}$$
        \end{enumerate}
      \end{block}
    \end{column}

    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Interconexão dos Subsistemas}
        \footnotesize
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \textbf{\texttt{instr\_mem}:} Fornece a instrução lida;
          \item \textbf{\texttt{control\_unit}:} Gera todos os sinais de controle;
          \item \textbf{\texttt{datapath}:} Conecta PC, somadores, RegFile, ULA e árvore de multiplexação;
          \item \textbf{\texttt{data\_mem}:} Executa operações de carga e armazenamento;
          \item \textbf{Observabilidade Externa:} Sinais essenciais de depuração expostos nas portas do Top-Level.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
