\begin{frame}{Objetivos da Segunda Entrega}
  \begin{block}{Escopo do Projeto}
    Implementação e validação completa do processador \textbf{MIPS Monociclo de 32 bits} em Verilog HDL, sintetizável para FPGA.
  \end{block}

  \vspace{0.4cm}
  \textbf{Principais Metas desta Entrega:}
  \begin{itemize}\setlength{\itemsep}{8pt}
    \item \textbf{Caminho de Dados com Saltos:} Suporte completo a desvios condicionais (\texttt{beq}) e incondicionais (\texttt{j}, \texttt{jal}, \texttt{jr}).
    \item \textbf{Síntese em FPGA:} Prototipagem na placa \textbf{DE10-Lite} (Intel MAX 10 \texttt{10M50DAF484C7G}).
    \item \textbf{Simulação RTL:} Validação comportamental de todos os módulos via Icarus Verilog e GTKWave.
    \item \textbf{Código 100\% Parametrizado:} Flexibilidade de barramentos de dados e endereços.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 3: Arquitetura Geral do Processador
% ------------------------------------------------------------------------------
\begin{frame}{Visão Geral da Arquitetura}
  O processador é estruturado em \textbf{4 grandes subsistemas} interconectados:

  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Caminho de Dados (Datapath)}
        \small
        Executa as instruções: PC, somadores, multiplexadores, Banco de Registradores e ULA.
      \end{block}
      \vspace{0.2cm}
      \begin{block}{2. Unidade de Controle}
        \small
        Decodifica \texttt{opcode} e \texttt{funct}, orquestrando os sinais de controle do hardware.
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{3. Memória de Instruções (ROM)}
        \small
        Armazena o código de máquina do programa a ser executado; leitura combinacional.
      \end{exampleblock}
      \vspace{0.2cm}
      \begin{exampleblock}{4. Memória de Dados (RAM)}
        \small
        Armazena variáveis em tempo de execução; leitura assíncrona (\texttt{lw}) e escrita síncrona (\texttt{sw}).
      \end{exampleblock}
    \end{column}
  \end{columns}

  \vspace{0.3cm}
  \centering
  \footnotesize \textbf{Modelo Harvard:} Barramentos separados para código e dados evitam conflitos de acesso no monociclo.
\end{frame}

% ------------------------------------------------------------------------------
% Slide 4: O Caminho de Dados (Datapath)
% ------------------------------------------------------------------------------
\begin{frame}{O Caminho de Dados (Datapath)}
  \begin{block}{Função Principal}
    Roteamento de dados entre registradores, operadores aritméticos e memórias em um único ciclo de clock.
  \end{block}

  \vspace{0.4cm}
  \textbf{Componentes Fundamentais:}
  \begin{itemize}\setlength{\itemsep}{6pt}
    \item \textbf{PC (Program Counter):} Registrador de 32 bits atualizado a cada borda de subida do clock.
    \item \textbf{Somador do PC:} Calcula sequencialmente $\text{PC} + 4$ para a próxima instrução.
    \item \textbf{Extensor de Sinal:} Converte imediatos de 16 bits para 32 bits com preservação de sinal.
    \item \textbf{Somador de Branch:} Calcula o endereço alvo: $(\text{PC}+4) + (\text{offset} \ll 2)$.
    \item \textbf{Multiplexadores em Cascata:} Decidem o endereço do próximo PC e as fontes de dados da ULA e registradores.
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 5: Suporte a Instruções de Desvio (Jumps)
% ------------------------------------------------------------------------------
\begin{frame}{Suporte aos Saltos Incondicionais}
  O processador implementa os 3 modos de salto da arquitetura MIPS:

  \vspace{0.3cm}
  \begin{itemize}\setlength{\itemsep}{9pt}
    \item \textbf{\texttt{j target} (Jump Incondicional):}
      \begin{itemize}
        \item Salto direto para endereço pseudo-absoluto.
        \item Equação de endereço: $\text{PC} \leftarrow \{\text{PC}+4[31:28],\; \text{target}[25:0],\; \text{2'b00}\}$.
      \end{itemize}

    \item \textbf{\texttt{jal target} (Jump and Link):}
      \begin{itemize}
        \item Chamada de função: executa o salto e salva o endereço de retorno.
        \item Salva $\text{PC}+4$ no registrador especial \textbf{\$ra (\$31)}.
      \end{itemize}

    \item \textbf{\texttt{jr \$rs} (Jump Register):}
      \begin{itemize}
        \item Retorno de procedimento carregando o endereço contido em registrador.
        \item Equação de endereço: $\text{PC} \leftarrow \text{Reg}[rs]$ (normalmente \texttt{jr \$ra}).
      \end{itemize}
  \end{itemize}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 6: Lógica de Seleção do Próximo PC
% ------------------------------------------------------------------------------
\begin{frame}{Lógica de Seleção do Próximo PC}
  Para suportar sequenciamento normal, branch e jumps no mesmo ciclo, foi utilizada uma \textbf{cascata de multiplexadores 2:1}:

  \vspace{0.4cm}
  \begin{enumerate}\setlength{\itemsep}{8pt}
    \item \textbf{1º Estágio (Branch):} \\
      Seleciona entre $\text{PC}+4$ e $\text{PC}_{\text{branch}}$ via controle:
      $$\text{pc\_src} = \text{branch} \;\&\; \text{zero}$$

    \item \textbf{2º Estágio (Jump Direto e JAL):} \\
      Seleciona entre o resultado pós-branch e $\text{PC}_{\text{jump}}$ via sinal \textbf{\texttt{jump}}.

    \item \textbf{3º Estágio (Jump Register - JR):} \\
      Seleciona entre o resultado anterior e o dado do registrador \textbf{\texttt{read\_data1}} via sinal \textbf{\texttt{jr}}.
  \end{enumerate}

  \vspace{0.2cm}
  \alert{\textbf{Hierarquia:}} O salto indireto (\texttt{jr}) possui a prioridade final no fechamento do ciclo.
\end{frame}

% ------------------------------------------------------------------------------
% Slide 7: Unidade Lógica e Aritmética (ULA)
% ------------------------------------------------------------------------------
\begin{frame}{Unidade Lógica e Aritmética (ULA)}
  \begin{block}{Características da ULA de 32 bits}
    Módulo combinacional parametrizado responsável por todas as operações de dados.
  \end{block}

  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \textbf{Operações Implementadas:}
      \begin{itemize}\setlength{\itemsep}{5pt}
        \item Lógicas: \texttt{AND}, \texttt{OR}, \texttt{NOR}.
        \item Aritméticas: Adição (\texttt{add}) e Subtração (\texttt{sub}).
        \item Comparação: \texttt{SLT} (\textit{Set on Less Than}) com tratamento de sinal via \texttt{\$signed}.
      \end{itemize}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Destaques de Projeto}
        \begin{itemize}\setlength{\itemsep}{5pt}
          \item \textbf{Somador Compartilhado:} \\
            Subtração calculada como $A + \bar{B} + 1$, poupando área física na FPGA.
          \item \textbf{Flag Zero Rápida:} \\
            Detecção por redução unária:
            $$\text{zero} = \sim|\,\text{alu\_result}$$
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 8: Unidade de Controle e Decodificação
% ------------------------------------------------------------------------------
\begin{frame}{Unidade de Controle Integrada}
  Orquestra o caminho de dados dividindo a tarefa em dois blocos combinacionais:

  \vspace{0.3cm}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{main\_control}}
        Decodifica o \texttt{opcode} de 6 bits:
        \begin{itemize}
          \item Define sinais para Tipo-R, \texttt{lw}, \texttt{sw}, \texttt{beq}, \texttt{j} e \texttt{jal}.
          \item Controla \texttt{reg\_dst}, \texttt{alu\_src}, \texttt{mem\_to\_reg}, etc.
        \end{itemize}
      \end{block}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{block}{\texttt{alu\_control}}
        Decodifica o campo \texttt{funct}:
        \begin{itemize}
          \item Gera o controle de 4 bits \texttt{alu\_ctrl} para a ULA.
          \item Detecta a instrução de retorno \textbf{\texttt{jr}} (\texttt{funct = 6'b001000}).
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.4cm}
  \begin{alertblock}{Trava de Segurança Crítica do JR}
    \centering
    \texttt{assign reg\_write = reg\_write\_wire \& \textasciitilde jr;} \\
    \small Impede que a instrução \texttt{jr} sobrescreva o registrador \textbf{\$ra} com lixo da ULA.
  \end{alertblock}
\end{frame}

% ------------------------------------------------------------------------------
% Slide 9: Banco de Registradores e Memórias
% ------------------------------------------------------------------------------
\begin{frame}{Banco de Registradores e Memórias}
  \begin{itemize}\setlength{\itemsep}{9pt}
    \item \textbf{Banco de Registradores (\texttt{reg\_file}):}
      \begin{itemize}
        \item 32 registradores de 32 bits com 2 portas de leitura combinacionais e 1 de escrita síncrona.
        \item \textbf{Regra MIPS:} Registrador \textbf{\$0 (\$zero)} é permanentemente travado em zero.
      \end{itemize}

    \item \textbf{Memória de Instruções (\texttt{instr\_mem} -- ROM):}
      \begin{itemize}
        \item Leitura por palavra indexada por \textbf{\texttt{PC[31:2]}} (divisão por 4 sem custo em hardware).
        \item Carga do código de máquina via diretiva \texttt{\$readmemh}.
      \end{itemize}

    \item \textbf{Memória de Dados (\texttt{data\_mem} -- RAM):}
      \begin{itemize}
        \item Leitura combinacional para \texttt{lw} e escrita síncrona habilitada por \texttt{mem\_write} para \texttt{sw}.
      \end{itemize}
  \end{itemize}
\end{frame}
